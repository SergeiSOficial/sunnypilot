function(python_extension target package output)
  target_include_directories(${target} PRIVATE ${PYTHON_ROOT}/include/python3.12)
  target_compile_options(${target} PRIVATE -Wno-error)
  target_link_libraries(${target} PRIVATE ${PYTHON_ROOT}/jniLibs/arm64-v8a/libpython3.12.so)
  set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "${output}" SUFFIX ".cpython-312.so"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/../../../build/native-python/${package}")
endfunction()

set(ORT_ROOT "${NATIVE_ROOT}/onnxruntime-1.31.1")

  message(FATAL_ERROR "Run android/scripts/prepare_native.py to build portable ONNX Runtime")
endif()
file(READ "${ORT_ROOT}/build.json" ORT_BUILD)
string(JSON ORT_OPTIONAL_KERNELS GET "${ORT_BUILD}" optional_sme_sve_kernels)
string(JSON ORT_EXPECTED_SHA GET "${ORT_BUILD}" library_sha256)
file(SHA256 "${ORT_ROOT}/jni/arm64-v8a/libonnxruntime.so" ORT_ACTUAL_SHA)
if(ORT_OPTIONAL_KERNELS OR NOT ORT_EXPECTED_SHA STREQUAL ORT_ACTUAL_SHA)
  message(FATAL_ERROR "Portable ONNX Runtime library does not match its build manifest")
endif()
